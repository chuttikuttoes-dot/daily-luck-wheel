        // Segment Configurations
        const segments = [
            { label: "💖 Love", color: "#ff4b82", category: "Love & Relationships" },
            { label: "💰 Wealth", color: "#ffd700", category: "Financial Opportunity" },
            { label: "🚀 Career", color: "#9b51e0", category: "Career Breakthrough" },
            { label: "⚡ Vitality", color: "#ff7d00", category: "Energy & Stamina" },
            { label: "🎨 Flow", color: "#00f2fe", category: "Creative Flow" },
            { label: "🔮 Wildcard", color: "#e040fb", category: "Cosmic Surprise" },

        const canvas = document.getElementById('wheel');
        const ctx = canvas.getContext('2d');
        const spinBtn = document.getElementById('spinBtn');
        const birthdayInput = document.getElementById('birthday');

        let currentAngle = 0;
        let isSpinning = false;
        let lastSegmentIndex = -1;

        // --- WEB AUDIO SYNTHESIZER ---
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
        }

        function playTickSound() {
            if (!audioCtx) return;
            try {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();

                osc.type = 'triangle';
                osc.frequency.setValueAtTime(480 + Math.random() * 40, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(120, audioCtx.currentTime + 0.03);

                gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.03);

                osc.connect(gain);
                gain.connect(audioCtx.destination);

                osc.start();
                osc.stop(audioCtx.currentTime + 0.03);
            } catch (e) {}
        }

        function playWinSound() {
            if (!audioCtx) return;
            try {
                const notes = [523.25, 659.25, 783.99, 1046.50]; // C5, E5, G5, C6 arpeggio
                notes.forEach((freq, index) => {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();

                    osc.type = 'sine';
                    osc.frequency.value = freq;

                    const startTime = audioCtx.currentTime + index * 0.12;
                    const duration = 0.4;

                    gain.gain.setValueAtTime(0.2, startTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, startTime + duration);

                    osc.connect(gain);
                    gain.connect(audioCtx.destination);

                    osc.start(startTime);
                    osc.stop(startTime + duration);
                });
            } catch (e) {}
        }

        // Draw Canvas Wheel
        function drawWheel() {
            const numSegments = segments.length;
            const arcSize = (2 * Math.PI) / numSegments;
            const centerX = canvas.width / 2;
            const centerY = canvas.height / 2;
            const radius = centerX - 10;

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            for (let i = 0; i < numSegments; i++) {
                const angle = currentAngle + i * arcSize;

                // Slice background
                ctx.beginPath();
                ctx.fillStyle = segments[i].color;
                ctx.moveTo(centerX, centerY);
                ctx.arc(centerX, centerY, radius, angle, angle + arcSize);
                ctx.lineTo(centerX, centerY);
                ctx.fill();
                ctx.strokeStyle = "rgba(11, 13, 27, 0.5)";
                ctx.lineWidth = 3;
                ctx.stroke();

                // Text labels
                ctx.save();
                ctx.translate(centerX, centerY);
                ctx.rotate(angle + arcSize / 2);
                ctx.textAlign = "right";
                ctx.fillStyle = "#ffffff";
                ctx.font = "bold 14px 'Segoe UI', sans-serif";
                ctx.shadowColor = "rgba(0,0,0,0.8)";
                ctx.shadowBlur = 4;
                ctx.fillText(segments[i].label, radius - 15, 5);
                ctx.restore();
            }

            // Center Pin
            ctx.beginPath();
            ctx.arc(centerX, centerY, 25, 0, 2 * Math.PI);
            ctx.fillStyle = "#0f1123";
            ctx.fill();
            ctx.strokeStyle = "#ffd700";
            ctx.lineWidth = 3;
            ctx.stroke();
        }

        drawWheel();

        // Deterministic Daily Hash
        function getDailyHash(birthdayStr) {
            const todayStr = new Date().toISOString().slice(0, 10);
            const combined = birthdayStr + todayStr;
            let hash = 0;
            for (let i = 0; i < combined.length; i++) {
                hash = (hash << 5) - hash + combined.charCodeAt(i);
                hash |= 0;
            }
            return Math.abs(hash);
        }

        // Zodiac & Numerology
        function getZodiacInfo(dateStr) {
            const date = new Date(dateStr);
            const day = date.getUTCDate();
            const month = date.getUTCMonth() + 1;
            const year = date.getUTCFullYear();

            let sign = "";
            let element = "";

            if ((month == 1 && day >= 20) || (month == 2 && day <= 18)) { sign = "Aquarius ♒"; element = "Air"; }
            else if ((month == 2 && day >= 19) || (month == 3 && day <= 20)) { sign = "Pisces ♓"; element = "Water"; }
            else if ((month == 3 && day >= 21) || (month == 4 && day <= 19)) { sign = "Aries ♈"; element = "Fire"; }
            else if ((month == 4 && day >= 20) || (month == 5 && day <= 20)) { sign = "Taurus ♉"; element = "Earth"; }
            else if ((month == 5 && day >= 21) || (month == 6 && day <= 20)) { sign = "Gemini ♊"; element = "Air"; }
            else if ((month == 6 && day >= 21) || (month == 7 && day <= 22)) { sign = "Cancer ♋"; element = "Water"; }
            else if ((month == 7 && day >= 23) || (month == 8 && day <= 22)) { sign = "Leo ♌"; element = "Fire"; }
            else if ((month == 8 && day >= 23) || (month == 9 && day <= 22)) { sign = "Virgo ♍"; element = "Earth"; }
            else if ((month == 9 && day >= 23) || (month == 10 && day <= 22)) { sign = "Libra ♎"; element = "Air"; }
            else if ((month == 10 && day >= 23) || (month == 11 && day <= 21)) { sign = "Scorpio ♏"; element = "Water"; }
            else if ((month == 11 && day >= 22) || (month == 12 && day <= 21)) { sign = "Sagittarius ♐"; element = "Fire"; }
            else { sign = "Capricorn ♑"; element = "Earth"; }

            let digits = (day.toString() + month.toString() + year.toString()).split('');
            let sum = digits.reduce((a, b) => parseInt(a) + parseInt(b), 0);
            while (sum > 9 && sum !== 11 && sum !== 22 && sum !== 33) {
                sum = sum.toString().split('').reduce((a, b) => parseInt(a) + parseInt(b), 0);
            }

            return { sign, element, lifePath: sum };
        }

        // Spin Handler
        spinBtn.addEventListener('click', () => {
            const bdayVal = birthdayInput.value;
            if (!bdayVal || isSpinning) return;

            initAudio(); // Activate browser Web Audio context

            isSpinning = true;
            spinBtn.disabled = true;

            const zInfo = getZodiacInfo(bdayVal);
            document.getElementById('zodiacSign').innerText = zInfo.sign;
            document.getElementById('zodiacElement').innerText = zInfo.element;
            document.getElementById('lifePath').innerText = zInfo.lifePath;
            document.getElementById('zodiacInfo').style.display = 'grid';

            const hash = getDailyHash(bdayVal);
            const targetSegmentIndex = hash % segments.length;
            const luckScore = 60 + (hash % 41);

            const numSegments = segments.length;
            const arcSize = (2 * Math.PI) / numSegments;
            
            const targetAngleOffset = (numSegments - targetSegmentIndex - 0.5) * arcSize;
            const extraRotations = 5 * 2 * Math.PI;
            const finalAngle = extraRotations + targetAngleOffset;

            let startTime = null;
            const duration = 4000;
            lastSegmentIndex = -1;

            function animateSpin(timestamp) {
                if (!startTime) startTime = timestamp;
                const progress = Math.min((timestamp - startTime) / duration, 1);
                
                const easeOut = 1 - Math.pow(1 - progress, 3);
                currentAngle = easeOut * finalAngle;

                // Track top pointer tick audio
                const pointerAngle = (1.5 * Math.PI - (currentAngle % (2 * Math.PI)) + 4 * Math.PI) % (2 * Math.PI);
                const currentSegment = Math.floor(pointerAngle / arcSize);

                if (currentSegment !== lastSegmentIndex) {
                    playTickSound();
                    lastSegmentIndex = currentSegment;
                }

                drawWheel();

                if (progress < 1) {
                    requestAnimationFrame(animateSpin);
                } else {
                    isSpinning = false;
                    spinBtn.disabled = false;
                    playWinSound(); // Victory audio
                    showResult(segments[targetSegmentIndex], luckScore, zInfo, hash);
                }
            }

            requestAnimationFrame(animateSpin);
        });

        // Display Fortune Result

            const resultCard = document.getElementById('resultCard');
            resultCard.style.display = 'block';

            document.getElementById('resultTitle').innerText = `${segment.label} (${segment.category})`;
            document.getElementById('luckFill').style.width = `${score}%`;
            document.getElementById('luckPercentage').innerText = `Daily Alignment: ${score}%`;

            const fortunes = [
                `Today your ${zodiac.element} energy naturally harmonizes with ${segment.category}. Trust your intuitive pulse around mid-day.`,
                `Cosmic alignment reveals a powerful spark in ${segment.category}. Your Life Path ${zodiac.lifePath} amplifies your focus today.`,
                `Celestial movements bring major momentum to ${segment.category}. Expect a pleasant turn of events before sunset!`
            ];
            document.getElementById('fortuneText').innerText = fortunes[hash % fortunes.length];

            const colors = ["Cosmic Gold", "Electric Cyan", "Royal Purple", "Emerald Green", "Solar Crimson"];
            document.getElementById('luckyNums').innerText = `${(hash % 9) + 1}, ${(hash % 23) + 7}`;
            document.getElementById('powerHour').innerText = `${(hash % 12) + 1}:00 ${(hash % 2 === 0 ? 'AM' : 'PM')}`;
            document.getElementById('luckyColor').innerText = colors[hash % colors.length];
            document.getElementById('auraFocus').innerText = zodiac.element;
        }
